-- B4a: triggers on the ACC config tables, BOX_ACC sequences, and where the PK generator lives
SELECT 'TRIGGER' AS kind, t.TABLE_NAME AS where_, t.TRIGGER_NAME AS name, t.TRIGGER_TYPE AS detail, t.STATUS AS extra
FROM   ALL_TRIGGERS t
WHERE  t.TABLE_OWNER = 'BOX_ACC'
AND    t.TABLE_NAME IN ('T_BOX_CONDPAR_PROP_S', 'T_BOX_CONF_LO_PROP_S', 'T_BOX_LINK_ARRAY_X', 'T_BOX_CROSS_ACCTCONF_S',
                        'T_BOX_ACCT_PORT_PROP_S', 'T_BOX_ACCT_LIST_TOPIC_S', 'T_BOX_ACCT_LIST_COND_S',
                        'T_BOX_ACCT_TOPICS_S', 'T_BOX_ACCT_HISTSTD', 'T_BOX_ACCT_GLTA_S', 'T_BOX_ACCT_PARENT_DATE_S',
                        'T_BOX_NETCONTRACT_S', 'T_BOX_CONF_INSTRUM_TYPE_S', 'T_BOX_NETCCS_BSPLAN_S',
                        'T_BOX_GLTA_LOCAL_S', 'T___SEQUENCE')
UNION ALL
SELECT 'SEQUENCE', s.SEQUENCE_OWNER, s.SEQUENCE_NAME, TO_CHAR(s.INCREMENT_BY), TO_CHAR(s.LAST_NUMBER)
FROM   ALL_SEQUENCES s WHERE s.SEQUENCE_OWNER = 'BOX_ACC'
UNION ALL
SELECT 'OBJECT', o.OWNER, o.OBJECT_NAME, o.OBJECT_TYPE, o.STATUS
FROM   ALL_OBJECTS o WHERE o.OBJECT_NAME IN ('F___SEQUENCE', 'T___SEQUENCE')
ORDER  BY 1, 2, 3;

-- B4b: which tables BOX_ACC's PK counter serves, and their next value
SELECT PK_NAME, NEXT_VAL FROM BOX_ACC.T___SEQUENCE ORDER BY PK_NAME;
