SELECT TRUNC(f.DDATE) RECON_DATE, l.FRONT_ID, f.*
FROM   DEVENG.T_PGT_IRFINANCST_S f
JOIN   (SELECT DISTINCT DEAL_ID, FRONT_ID FROM DEVENG.T_PGT_IRDATAMIS_S
        WHERE TRUNC(DDATE) = DATE '2026-09-25') l ON l.DEAL_ID = f.DEAL_ID
WHERE  TRUNC(f.DDATE) = DATE '2026-09-25' AND f.FK_BRANCH = 20007.4
ORDER  BY l.FRONT_ID, f.FK_DIRECTION, f.FK_CURRENCY;
