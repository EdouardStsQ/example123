SELECT 'DATADEAL' step, SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1) front_id, COUNT(*) n
FROM   BOX_FE.T_BOX_DATADEAL_S
WHERE  DEAL_ID IN ('76278766.513','79333922.513','79760689.513','80855116.513','80855318.513')
GROUP  BY SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1)
UNION ALL
SELECT 'IRFINANCST', SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1), COUNT(*)
FROM   BOX_FE.T_BOX_IRFINANCST_S
WHERE  DEAL_ID IN ('76278766.513','79333922.513','79760689.513','80855116.513','80855318.513')
GROUP  BY SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1)
ORDER  BY 1, 2;
