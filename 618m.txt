SELECT DEAL_ID, DDATE, FK_DIRECTION, NPVMAN, NPVACC
FROM   BOX_FE.T_BOX_MTM_DATA_S
WHERE  DEAL_ID IN (76278766.513, 79333922.513, 79760689.513, 80855116.513, 80855318.513)
ORDER  BY DEAL_ID, DDATE;

SELECT FRONT_ID, INSTRUMENT, TRUNC(PROCESSDATE) pdate, COUNT(*) n
FROM   BOX_FE.T_BOX_RAW_MARKET_DATA_S
WHERE  FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318')
GROUP  BY FRONT_ID, INSTRUMENT, TRUNC(PROCESSDATE)
ORDER  BY 1, 3;
