SELECT r.BOOK,
       COUNT(DISTINCT r.FRONT_ID) N_RAW,
       COUNT(DISTINCT d.FRONT_ID) N_DEAL_DATA,
       COUNT(DISTINCT CASE WHEN f.DEAL_ID IS NOT NULL THEN d.FRONT_ID END) N_FINANCST
FROM   BOX_FE.T_BOX_RAW_DEAL_DATA_S r
LEFT   JOIN BOX_FE.T_BOX_DEAL_DATA_S d ON d.FRONT_ID = r.FRONT_ID
LEFT   JOIN (SELECT DISTINCT DEAL_ID FROM BOX_FE.T_BOX_IRFINANCST_S
             WHERE TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4) f ON f.DEAL_ID = d.DEAL_ID
WHERE  TRUNC(r.PROCESSDATE) = DATE '2026-09-25' AND r.INSTRUMENT = 'IRS'
GROUP  BY r.BOOK ORDER BY 1;

-- B5
SELECT N_LEGS_DEAL_DATA, FIN_DIRECTION, COUNT(*) N_DEALS
FROM  (SELECT f.DEAL_ID, MAX(f.FK_DIRECTION) FIN_DIRECTION,
              (SELECT COUNT(*) FROM BOX_FE.T_BOX_DEAL_DATA_S d WHERE d.DEAL_ID = f.DEAL_ID) N_LEGS_DEAL_DATA
       FROM   BOX_FE.T_BOX_IRFINANCST_S f
       WHERE  TRUNC(f.DDATE) = DATE '2026-09-25' AND f.FK_BRANCH = 20007.4
       GROUP  BY f.DEAL_ID HAVING COUNT(*) = 1)
GROUP  BY N_LEGS_DEAL_DATA, FIN_DIRECTION ORDER BY 3 DESC;

-- G7
SELECT CASE WHEN d.MATURITYDATE < DATE '2026-09-25' THEN 'matured before 25-Sep' ELSE 'live on 25-Sep' END MATURITY,
       CASE WHEN f.DEAD_DATE IS NULL THEN 'no DEAD_DATE' ELSE 'DEAD_DATE set' END DEAD,
       COUNT(DISTINCT f.DEAL_ID) N_DEALS
FROM   DEVENG.T_PGT_IRFINANCST_S f
JOIN   DEVENG.T_PGT_IRDATAMIS_S d ON d.DEAL_ID = f.DEAL_ID AND TRUNC(d.DDATE) = DATE '2026-09-25'
WHERE  TRUNC(f.DDATE) = DATE '2026-09-25' AND f.FK_BRANCH = 20007.4
GROUP  BY CASE WHEN d.MATURITYDATE < DATE '2026-09-25' THEN 'matured before 25-Sep' ELSE 'live on 25-Sep' END,
          CASE WHEN f.DEAD_DATE IS NULL THEN 'no DEAD_DATE' ELSE 'DEAD_DATE set' END;
