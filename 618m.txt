SELECT PK, ALIASCODE, FK_PARENT, FK_OWNER_OBJ, FK_EXTENSION, FK_SOURCE
FROM   PGT_TRD.T_PGT_OBJ_SOURCE_S
WHERE  UPPER(ALIASCODE) = 'TR_AL_AFS_NY';


--
SELECT column_name FROM all_tab_columns
WHERE  owner = 'PGT_TRD' AND table_name = 'T_PGT_FOLDER_S' ORDER BY column_id;


--
SELECT PK, SHORTNAME, DESCRIPTION, FK_BRANCH
FROM   PGT_TRD.T_PGT_FOLDER_S
WHERE  UPPER(SHORTNAME) = 'TR_AL_AFS_NY' OR UPPER(DESCRIPTION) LIKE '%TR_AL_AFS_NY%';


--
SELECT s.FK_SOURCE, COUNT(DISTINCT f.PK) n_folders
FROM   PGT_TRD.T_PGT_FOLDER_S f
LEFT   JOIN PGT_TRD.T_PGT_OBJ_SOURCE_S s ON s.FK_PARENT = f.PK
       AND s.FK_OWNER_OBJ = 1453.4 AND s.FK_EXTENSION = 10513.4
WHERE  f.FK_BRANCH = 20007.4
GROUP  BY s.FK_SOURCE ORDER BY 1;
