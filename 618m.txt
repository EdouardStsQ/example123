-- label aliases (owner 17910.4, list 118615.4): which sources, how many
SELECT a.FK_SOURCE, s.SHORTNAME, s.DESCRIPTION, COUNT(*) AS n_aliases
FROM   PGT_TRD.T_PGT_OBJ_SOURCE_S a
LEFT   JOIN PGT_SYS.T_PGT_SOURCE_S s ON s.PK = a.FK_SOURCE
WHERE  a.FK_OWNER_OBJ = 17910.4 AND a.FK_EXTENSION = 118615.4
GROUP  BY a.FK_SOURCE, s.SHORTNAME, s.DESCRIPTION ORDER BY n_aliases DESC;
