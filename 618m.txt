-- B6 the single-row deals: legs and directions in RAW
SELECT N_RAW_ROWS, N_RAW_DIRECTIONS, COUNT(*) N_TRADES
FROM  (SELECT d.FRONT_ID, COUNT(r.FRONT_ID) N_RAW_ROWS, COUNT(DISTINCT r.DIRECTION) N_RAW_DIRECTIONS
       FROM  (SELECT DISTINCT dd.FRONT_ID FROM BOX_FE.T_BOX_DEAL_DATA_S dd
              WHERE dd.DEAL_ID IN (SELECT DEAL_ID FROM BOX_FE.T_BOX_IRFINANCST_S
                                   WHERE TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4
                                   GROUP BY DEAL_ID HAVING COUNT(*) = 1)) d
       LEFT  JOIN BOX_FE.T_BOX_RAW_DEAL_DATA_S r
              ON r.FRONT_ID = d.FRONT_ID AND TRUNC(r.PROCESSDATE) = DATE '2026-09-25'
       GROUP BY d.FRONT_ID)
GROUP  BY N_RAW_ROWS, N_RAW_DIRECTIONS ORDER BY 3 DESC;

-- B7 the RAW trades the import dropped: execution type and maturity, per book
SELECT r.BOOK, r.DATA_EXECUTION_TYPE,
       CASE WHEN r.MATURITYDATE < DATE '2026-09-25' THEN 'matured' ELSE 'live' END MATURITY,
       COUNT(DISTINCT r.FRONT_ID) N_TRADES
FROM   BOX_FE.T_BOX_RAW_DEAL_DATA_S r
WHERE  TRUNC(r.PROCESSDATE) = DATE '2026-09-25' AND r.INSTRUMENT = 'IRS'
AND    NOT EXISTS (SELECT 1 FROM BOX_FE.T_BOX_DEAL_DATA_S d WHERE d.FRONT_ID = r.FRONT_ID)
GROUP  BY r.BOOK, r.DATA_EXECUTION_TYPE,
          CASE WHEN r.MATURITYDATE < DATE '2026-09-25' THEN 'matured' ELSE 'live' END
ORDER  BY 1, 2, 3;
