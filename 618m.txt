SELECT 'DEAL_DATA' tbl, FRONT_ID, COUNT(*) n FROM BOX_FE.T_BOX_DEAL_DATA_S
WHERE  FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318') GROUP BY FRONT_ID
UNION ALL
SELECT 'FLOW_DATA', FRONT_ID, COUNT(*) FROM BOX_FE.T_BOX_FLOW_DATA_S
WHERE  FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318') GROUP BY FRONT_ID
ORDER  BY 1, 2;

WITH r AS (SELECT DISTINCT FRONT_ID, SOURCESYSTEM, FOLDER, COUNTERPARTY, CURRENCY, INSTRUMENT
           FROM BOX_FE.T_BOX_RAW_DEAL_DATA_S
           WHERE FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318')
           AND TRUNC(PROCESSDATE) = DATE '2026-09-24')
SELECT 'source system' field, r.FRONT_ID, r.SOURCESYSTEM raw_value, TO_CHAR(s.FK_PARENT) static_pk,
       CASE WHEN s.ALIASCODE IS NULL THEN 'MISSING' ELSE 'OK' END status
FROM r LEFT JOIN PGT_STC.T_PGT_OBJSRC_INPUT_S s ON s.ALIASCODE = r.SOURCESYSTEM
       AND s.FK_OWNER_OBJ = 1341.4 AND s.FK_EXTENSION = 110862.4 AND s.FK_SOURCE = 577.4
UNION ALL
SELECT 'folder', r.FRONT_ID, r.FOLDER, TO_CHAR(s.FK_PARENT), CASE WHEN s.PK IS NULL THEN 'MISSING' ELSE 'OK' END
FROM r LEFT JOIN PGT_TRD.T_PGT_OBJ_SOURCE_S s ON s.ALIASCODE = r.FOLDER
       AND s.FK_OWNER_OBJ = 1453.4 AND s.FK_EXTENSION = 10513.4 AND s.FK_SOURCE = 279.4
UNION ALL
SELECT 'counterparty', r.FRONT_ID, r.COUNTERPARTY, TO_CHAR(s.FK_PARENT), CASE WHEN s.PK IS NULL THEN 'MISSING' ELSE 'OK' END
FROM r LEFT JOIN PGT_TRD.T_PGT_OBJ_SOURCE_S s ON s.ALIASCODE = r.COUNTERPARTY
       AND s.FK_OWNER_OBJ = 1448.4 AND s.FK_EXTENSION = 10485.4 AND s.FK_SOURCE = 20.4
UNION ALL
SELECT 'currency', r.FRONT_ID, r.CURRENCY, TO_CHAR(s.FK_PARENT), CASE WHEN s.PK IS NULL THEN 'MISSING' ELSE 'OK' END
FROM r LEFT JOIN PGT_TRD.T_PGT_OBJ_SOURCE_S s ON s.ALIASCODE = r.CURRENCY
       AND s.FK_OWNER_OBJ = 1382.4 AND s.FK_EXTENSION = 10206.4 AND s.FK_SOURCE = 20.4
UNION ALL
SELECT 'instrument', r.FRONT_ID, r.INSTRUMENT, TO_CHAR(s.FK_PARENT), CASE WHEN s.PK IS NULL THEN 'MISSING' ELSE 'OK' END
FROM r LEFT JOIN PGT_TRD.T_PGT_OBJ_SOURCE_S s ON s.ALIASCODE = r.INSTRUMENT
       AND s.FK_OWNER_OBJ = 1395.4 AND s.FK_EXTENSION = 10243.4 AND s.FK_SOURCE = 586.4
ORDER BY 2, 1;

SELECT ALIASCODE, FK_OWNER_OBJ, FK_EXTENSION, FK_SOURCE, FK_PARENT FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
WHERE  ALIASCODE IN (:raw_value) AND FK_OWNER_OBJ IN (1453.4, 1448.4);

SELECT r.FRONT_ID, r.DIRECTION, r.BOOK, r.INSTRUMENT, r.DATA_EXECUTION_TYPE, r.MATURITYDATEREAL,
       CASE WHEN r.INSTRUMENT = 'IRS' THEN 'OK' ELSE 'CHECK' END instr_ok,
       CASE WHEN UPPER(r.DATA_EXECUTION_TYPE) = 'BOOK' THEN 'OK' ELSE 'FILTERED' END exec_type_ok,
       CASE WHEN TRUNC(r.MATURITYDATEREAL) + 5 >= DATE '2026-09-24' THEN 'OK' ELSE 'CHECK (matured)' END maturity_ok,
       f.FK_BRANCH folder_branch, CASE WHEN f.FK_BRANCH = 20007.4 THEN 'OK' ELSE 'CHECK' END folder_branch_ok
FROM   BOX_FE.T_BOX_RAW_DEAL_DATA_S r
LEFT   JOIN PGT_TRD.T_PGT_OBJ_SOURCE_S s ON s.ALIASCODE = r.FOLDER AND s.FK_OWNER_OBJ = 1453.4
       AND s.FK_EXTENSION = 10513.4 AND s.FK_SOURCE = 279.4
LEFT   JOIN PGT_TRD.T_PGT_FOLDER_S f ON f.PK = s.FK_PARENT
WHERE  r.FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318')
AND    TRUNC(r.PROCESSDATE) = DATE '2026-09-24';
