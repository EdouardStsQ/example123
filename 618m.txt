-- what parameter type 367.4 is
SELECT PK, DESCRIPTION FROM PGT_SYS.PGT_DOMAINS WHERE PK = 367.4;
-- the CCS instrument types and their 367.4 value: which ones the Tier 1 rule nets
SELECT it.PK, it.DESCRIPTION, NVL(p.PARAMETERVALUE, 'NO') AS p367
FROM   PGT_SYS.T_PGT_INSTRUMENT_TYPE_S it
LEFT   JOIN PGT_STC.T_PGT_PARAMETER_S p ON p.FK_PARENT = it.PK AND p.FK_PARAMETERTYPE = 367.4
WHERE  it.FK_PARENT = 20.4;
