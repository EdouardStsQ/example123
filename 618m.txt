-- G1 the columns
SELECT TABLE_NAME, COLUMN_ID, COLUMN_NAME, DATA_TYPE
FROM   ALL_TAB_COLUMNS
WHERE  OWNER = 'DEVENG' AND TABLE_NAME IN ('T_PGT_IRFINANCST_S', 'T_PGT_IRDATAMIS_S')
ORDER  BY 1, 2;

-- G2 rows per deal in the financial table alone (no join)
SELECT N_ROWS_PER_DEAL, COUNT(*) N_DEALS
FROM  (SELECT DEAL_ID, COUNT(*) N_ROWS_PER_DEAL
       FROM   DEVENG.T_PGT_IRFINANCST_S
       WHERE  TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4
       GROUP  BY DEAL_ID)
GROUP  BY N_ROWS_PER_DEAL ORDER BY 1;

-- G3 deals with several FRONT_IDs in the deal table
SELECT N_FRONT_IDS_PER_DEAL, COUNT(*) N_DEALS
FROM  (SELECT d.DEAL_ID, COUNT(DISTINCT d.FRONT_ID) N_FRONT_IDS_PER_DEAL
       FROM   DEVENG.T_PGT_IRDATAMIS_S d
       WHERE  d.DEAL_ID IN (SELECT DEAL_ID FROM DEVENG.T_PGT_IRFINANCST_S
                            WHERE TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4)
       GROUP  BY d.DEAL_ID)
GROUP  BY N_FRONT_IDS_PER_DEAL ORDER BY 1;
