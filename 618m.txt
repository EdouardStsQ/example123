SELECT COUNT(*) N_ROWS, COUNT(DISTINCT f.DEAL_ID) N_DEALS, COUNT(DISTINCT l.FRONT_ID) N_FRONT_IDS,
       SUM(CASE WHEN l.FRONT_ID IS NULL THEN 1 ELSE 0 END) N_NO_FRONT_ID
FROM   DEVENG.T_PGT_IRFINANCST_S f
LEFT   JOIN (SELECT DISTINCT DEAL_ID, FRONT_ID FROM DEVENG.T_PGT_IRDATAMIS_S) l ON l.DEAL_ID = f.DEAL_ID
WHERE  TRUNC(f.DDATE) = DATE '2026-09-25' AND f.FK_BRANCH = 20007.4;

SELECT COUNT(*) N_ROWS, COUNT(DISTINCT f.DEAL_ID) N_DEALS, COUNT(DISTINCT l.FRONT_ID) N_FRONT_IDS,
       SUM(CASE WHEN l.FRONT_ID IS NULL THEN 1 ELSE 0 END) N_NO_FRONT_ID
FROM   BOX_FE.T_BOX_IRFINANCST_S f
LEFT   JOIN (SELECT DISTINCT DEAL_ID, FRONT_ID FROM BOX_FE.T_BOX_DEAL_DATA_S) l ON l.DEAL_ID = f.DEAL_ID
WHERE  TRUNC(f.DDATE) = DATE '2026-09-25' AND f.FK_BRANCH = 20007.4;
