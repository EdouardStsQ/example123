-- G4 which columns make a row unique (the one equal to N_ROWS is the key)
SELECT COUNT(*) N_ROWS,
       COUNT(DISTINCT DEAL_ID||'|'||FK_DIRECTION)                     K_DIR,
       COUNT(DISTINCT DEAL_ID||'|'||FK_DIRECTION||'|'||FK_CURRENCY)   K_DIR_CCY,
       COUNT(DISTINCT DEAL_ID||'|'||FK_DIRECTION||'|'||FK_EXTENSION)  K_DIR_EXT,
       COUNT(DISTINCT DEAL_ID||'|'||FK_DIRECTION||'|'||CRITERIAL)     K_DIR_CRIT,
       COUNT(DISTINCT DEAL_ID||'|'||FK_DIRECTION||'|'||FK_FEETYPE)    K_DIR_FEE,
       COUNT(DISTINCT DEAL_ID||'|'||FK_INTRATE)                       K_INTRATE
FROM   DEVENG.T_PGT_IRFINANCST_S
WHERE  TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4;

-- G5 what the 4-row deals' rows look like
SELECT FK_DIRECTION, FK_EXTENSION, FK_CURRENCY, CRITERIAL, FK_FEETYPE, COUNT(*) N
FROM   DEVENG.T_PGT_IRFINANCST_S
WHERE  TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4
AND    DEAL_ID IN (SELECT DEAL_ID FROM DEVENG.T_PGT_IRFINANCST_S
                   WHERE TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4
                   GROUP BY DEAL_ID HAVING COUNT(*) = 4)
GROUP  BY FK_DIRECTION, FK_EXTENSION, FK_CURRENCY, CRITERIAL, FK_FEETYPE
ORDER  BY N DESC;

-- G6 FRONT_IDs per deal on 25-Sep only (0 = the deal has no deal-table row that day)
SELECT N_FRONT_IDS_PER_DEAL, COUNT(*) N_DEALS
FROM  (SELECT f.DEAL_ID, COUNT(DISTINCT d.FRONT_ID) N_FRONT_IDS_PER_DEAL
       FROM  (SELECT DISTINCT DEAL_ID FROM DEVENG.T_PGT_IRFINANCST_S
              WHERE TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4) f
       LEFT  JOIN DEVENG.T_PGT_IRDATAMIS_S d
              ON d.DEAL_ID = f.DEAL_ID AND TRUNC(d.DDATE) = DATE '2026-09-25'
       GROUP BY f.DEAL_ID)
GROUP  BY N_FRONT_IDS_PER_DEAL ORDER BY 1;
