-- J-E11 - the sequence (or counter) behind T_BOX_MBJ_PROPERTIES_S, and its highest PK with this installation's fraction
SELECT SEQUENCE_NAME, LAST_NUMBER, INCREMENT_BY, CACHE_SIZE FROM ALL_SEQUENCES
WHERE  SEQUENCE_OWNER = 'BOX_ACC' AND SEQUENCE_NAME LIKE '%MBJ%';
SELECT * FROM BOX_ACC.T___SEQUENCE WHERE PK_NAME = 'T_BOX_MBJ_PROPERTIES_S';
SELECT MAX(TRUNC(PK)) AS max_pk, COUNT(*) AS n FROM BOX_ACC.T_BOX_MBJ_PROPERTIES_S WHERE PK - TRUNC(PK) = 0.44;
