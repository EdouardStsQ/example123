SELECT FRONT_ID, DIRECTION, CURRENCY, TRUNC(PROCESSDATE) PROCESSDATE, COUNT(*) N_FLOWS,
       MIN(PAYMENTDATE) FIRST_PAYMENT, MAX(PAYMENTDATE) LAST_PAYMENT,
       SUM(CASE WHEN PAYMENTDATE <= DATE '2026-09-25' THEN AMOUNT ELSE 0 END) PAID_TO_DATE
FROM   BOX_FE.T_BOX_FLOW_DATA_S
WHERE  FRONT_ID IN ('33340980', '18934788', '37298510')
GROUP  BY FRONT_ID, DIRECTION, CURRENCY, TRUNC(PROCESSDATE)
ORDER  BY 1, 2, 3, 4;
