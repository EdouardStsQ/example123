SELECT column_id, column_name, data_type, nullable, data_default
FROM   all_tab_columns WHERE owner = 'PGT_TRD' AND table_name = 'T_PGT_OBJ_SOURCE_S' ORDER BY column_id;

-- 2. two existing folder aliases for the same source, as templates
SELECT * FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
WHERE  FK_OWNER_OBJ = 1453.4 AND FK_EXTENSION = 10513.4 AND FK_SOURCE IN (279.4, 513.4) AND ROWNUM <= 4;

-- 3. how SIGOM allocates PKs here (the function and its arguments)
SELECT owner, object_name, procedure_name FROM all_procedures
WHERE  object_name IN ('PKG_SIGOM_COV', 'F_GETPK', 'F___SEQUENCE')
   OR  procedure_name IN ('P_SETHEADER', 'F_GETPK')
ORDER  BY 1, 2, 3;
