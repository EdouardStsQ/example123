-- the counter SIGOM's F_GetPK may use, and the table's own sequence
SELECT * FROM BOX_ACC.T___SEQUENCE WHERE PK_NAME = 'T_BOX_ACCT_PARENT_DATE_S';
SELECT SEQUENCE_NAME, LAST_NUMBER, INCREMENT_BY, CACHE_SIZE FROM ALL_SEQUENCES
WHERE  SEQUENCE_OWNER = 'BOX_ACC' AND SEQUENCE_NAME LIKE '%PARENT_DATE%';
-- a trigger or code that sets the PK
SELECT TRIGGER_NAME, TRIGGERING_EVENT, STATUS FROM ALL_TRIGGERS
WHERE  TABLE_OWNER = 'BOX_ACC' AND TABLE_NAME = 'T_BOX_ACCT_PARENT_DATE_S';
SELECT OWNER, NAME, TYPE, LINE, TEXT FROM ALL_SOURCE WHERE UPPER(TEXT) LIKE '%T_BOX_ACCT_PARENT_DATE_S%';
-- what is already used, per PK fraction
SELECT PK - TRUNC(PK) AS fraction, COUNT(*), MAX(TRUNC(PK)) FROM BOX_ACC.T_BOX_ACCT_PARENT_DATE_S GROUP BY PK - TRUNC(PK);
