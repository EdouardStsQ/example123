SELECT 'RAW' SRC, r.FRONT_ID, r.DIRECTION, r.CURRENCY,
       CAST(NULL AS VARCHAR2(30)) CURRENCY2, CAST(NULL AS VARCHAR2(30)) DELIVERY_CURR, COUNT(*) N_ROWS
FROM   BOX_FE.T_BOX_RAW_DEAL_DATA_S r
WHERE  r.FRONT_ID IN ('10107111', '10568442', '12662656', '13934404', '14055968')
AND    TRUNC(r.PROCESSDATE) = DATE '2026-09-25'
GROUP  BY r.FRONT_ID, r.DIRECTION, r.CURRENCY
UNION ALL
SELECT 'DEAL_DATA', d.FRONT_ID, d.DIRECTION, d.CURRENCY, d.CURRENCY2, d.DELIVERY_CURR, COUNT(*)
FROM   BOX_FE.T_BOX_DEAL_DATA_S d
WHERE  d.FRONT_ID IN ('10107111', '10568442', '12662656', '13934404', '14055968')
GROUP  BY d.FRONT_ID, d.DIRECTION, d.CURRENCY, d.CURRENCY2, d.DELIVERY_CURR
ORDER  BY 2, 1, 3, 4;
