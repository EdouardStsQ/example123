-- 1. the sequence's next value in Tier 2
SELECT LAST_NUMBER, CACHE_SIZE FROM ALL_SEQUENCES
WHERE  SEQUENCE_OWNER = 'BOX_ACC' AND SEQUENCE_NAME = 'SPK_T_BOX_ACCT_PARENT_DATE_S';
-- 2. the rows already there, per PK fraction (compare with Tier 2's auth code fraction)
SELECT PK - TRUNC(PK) AS fraction, COUNT(*) AS n, MAX(TRUNC(PK)) AS max_pk
FROM   BOX_ACC.T_BOX_ACCT_PARENT_DATE_S GROUP BY PK - TRUNC(PK);
-- 3. whether SIGOM uses the counter instead of the sequence for this table
SELECT * FROM BOX_ACC.T___SEQUENCE WHERE PK_NAME = 'T_BOX_ACCT_PARENT_DATE_S';
