WITH b AS (SELECT LINE, TEXT FROM ALL_SOURCE
           WHERE OWNER = 'BOX_FE' AND NAME = 'PKG_FE_DEAL_CALCULATION' AND TYPE = 'PACKAGE BODY'),
     p AS (SELECT MIN(LINE) S FROM b WHERE UPPER(TEXT) LIKE '%PROCEDURE P_IMPORT_DEAL_DATA%'),
     e AS (SELECT MIN(b.LINE) E FROM b, p
           WHERE b.LINE > p.S AND (UPPER(b.TEXT) LIKE '%PROCEDURE %' OR UPPER(b.TEXT) LIKE '%FUNCTION %'))
SELECT b.LINE, b.TEXT
FROM   b, p, e
WHERE  b.LINE BETWEEN p.S AND NVL(e.E, p.S + 400)
ORDER  BY b.LINE;
