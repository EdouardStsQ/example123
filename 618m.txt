SET SERVEROUTPUT ON
DECLARE
  c_commit CONSTANT BOOLEAN      := FALSE;           -- TRUE only after a clean rehearsal
  c_alias  CONSTANT VARCHAR2(30) := 'TR_AL_AFS_NY';
  c_folder CONSTANT NUMBER       := 12.22;           -- TRALCOFISC, NY (as PRE)
  c_owner  CONSTANT NUMBER       := 1453.4;          -- folder alias
  c_ext    CONSTANT NUMBER       := 10513.4;
  c_src    CONSTANT NUMBER       := 279.4;           -- the import's source only
  v_pk NUMBER; v_own NUMBER; v_n PLS_INTEGER; v_auth NUMBER; v_frac NUMBER; v_code VARCHAR2(100);
BEGIN
  IF DBMS_TRANSACTION.LOCAL_TRANSACTION_ID IS NOT NULL THEN
    RAISE_APPLICATION_ERROR(-20000, 'Open transaction in this session: type ROLLBACK; and run again.');
  END IF;
  SELECT auth_code INTO v_auth FROM gom_glb_sys.t__core_info_s;
  v_frac := v_auth / POWER(10, LENGTH(TO_CHAR(v_auth)));
  DBMS_OUTPUT.PUT_LINE('installation fraction: ' || v_frac);

  SELECT COUNT(*) INTO v_n FROM PGT_STC.T_PGT_FOLDER_S
  WHERE  PK = c_folder AND CODE = 'TRALCOFISC' AND FK_BRANCH = 20007.4;
  IF v_n <> 1 THEN RAISE_APPLICATION_ERROR(-20001, 'Folder 12.22 TRALCOFISC (NY) not found. Nothing done.'); END IF;

  -- unique key 001: one alias per folder + owner + extension + source
  SELECT MAX(ALIASCODE), COUNT(*) INTO v_code, v_n FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
  WHERE  FK_PARENT = c_folder AND FK_OWNER_OBJ = c_owner AND FK_EXTENSION = c_ext AND FK_SOURCE = c_src;
  IF v_n > 0 THEN
    RAISE_APPLICATION_ERROR(-20004, 'Folder 12.22 already has alias ' || v_code || ' for source 279.4. Nothing done.');
  END IF;
  -- unique key 002: one alias code per owner + source
  SELECT COUNT(*) INTO v_n FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
  WHERE  FK_OWNER_OBJ = c_owner AND FK_SOURCE = c_src AND ALIASCODE = c_alias;
  IF v_n > 0 THEN
    RAISE_APPLICATION_ERROR(-20005, 'Alias ' || c_alias || ' already used for source 279.4 (another folder?). Nothing done.');
  END IF;

  v_pk  := NULL;                        -- IN/OUT: empty, so SIGOM gives a new PK
  v_own := c_owner;
  PGT_TRD.PKG_SIGOM_COV.P_SETHEADER('T_PGT_OBJ_SOURCE_S', v_pk, v_own);
  DBMS_OUTPUT.PUT_LINE('P_SETHEADER returned PK ' || v_pk || ', owner ' || v_own);
  IF v_pk IS NULL THEN RAISE_APPLICATION_ERROR(-20002, 'No PK returned. Nothing kept.'); END IF;
  SELECT COUNT(*) INTO v_n FROM PGT_TRD.T_PGT_OBJ_SOURCE_S WHERE PK = v_pk;
  IF v_n > 0 THEN RAISE_APPLICATION_ERROR(-20006, 'PK ' || v_pk || ' already used (counter behind). Nothing kept.'); END IF;

  INSERT INTO PGT_TRD.T_PGT_OBJ_SOURCE_S (PK, FK_OWNER_OBJ, FK_PARENT, FK_EXTENSION, ALIASCODE, FK_SOURCE)
  VALUES (v_pk, c_owner, c_folder, c_ext, c_alias, c_src);

  SELECT COUNT(*) INTO v_n FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
  WHERE  ALIASCODE = c_alias AND FK_OWNER_OBJ = c_owner AND FK_EXTENSION = c_ext
  AND    FK_PARENT = c_folder AND FK_SOURCE = c_src;
  IF v_n <> 1 THEN RAISE_APPLICATION_ERROR(-20003, v_n || ' row(s), expected 1. Nothing kept.'); END IF;
  DBMS_OUTPUT.PUT_LINE('OK - alias ' || c_alias || ' (source 279.4) -> folder 12.22, PK ' || v_pk);

  IF c_commit THEN COMMIT; DBMS_OUTPUT.PUT_LINE('COMMITTED');
  ELSE ROLLBACK; DBMS_OUTPUT.PUT_LINE('REHEARSAL - rolled back, nothing kept'); END IF;
EXCEPTION WHEN OTHERS THEN ROLLBACK; RAISE;
END;
/

SELECT PK, ALIASCODE, FK_PARENT, FK_SOURCE FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
WHERE  FK_PARENT = 12.22 AND FK_OWNER_OBJ = 1453.4 AND FK_EXTENSION = 10513.4
ORDER  BY FK_SOURCE;

DELETE FROM PGT_TRD.T_PGT_OBJ_SOURCE_S
WHERE  PK = <pk> AND ALIASCODE = 'TR_AL_AFS_NY' AND FK_PARENT = 12.22 AND FK_SOURCE = 279.4;
COMMIT;
