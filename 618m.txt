SELECT 'IRDATADEAL' tbl, SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1) front_id, COUNT(*) n
FROM   BOX_FE.T_BOX_IRDATADEAL_S
WHERE  DEAL_ID IN ('76278766.513','79333922.513','79760689.513','80855116.513','80855318.513')
GROUP  BY SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1)
UNION ALL
SELECT 'MTM_DATA', FRONT_ID, COUNT(*) FROM BOX_FE.T_BOX_MTM_DATA_S
WHERE  FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318') GROUP BY FRONT_ID
ORDER  BY 1, 2;
