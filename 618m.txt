SELECT LINE, TRIM(TEXT) AS TEXT
FROM   ALL_SOURCE
WHERE  OWNER = 'BOX_ACC' AND NAME = 'PKG_ACCTPRECOMMIT' AND TYPE = 'PACKAGE BODY'
AND   (UPPER(TEXT) LIKE '%PROCEDURE %' OR UPPER(TEXT) LIKE '%FUNCTION %'
       OR UPPER(TEXT) LIKE '%INSERT %' OR UPPER(TEXT) LIKE '%UPDATE %'
       OR UPPER(TEXT) LIKE '%DELETE %' OR UPPER(TEXT) LIKE '%MERGE %')
ORDER  BY LINE;
