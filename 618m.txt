SELECT PK, NOM_FUNC, DESCRIPTION
FROM   BOX_ACC.T_BOX_CONDPAR_PROP_S
WHERE  PK IN (12.65, 13.65);


SELECT c.FK_CONDITION, c.VALUE_FUNC, COUNT(*) AS n_props
FROM   BOX_ACC.T_BOX_ACCT_LIST_COND_S c
WHERE  c.FK_OWNER_OBJ = 35000170.65
AND    c.FK_EXTENSION = 35001514.65
AND    c.FK_CONDITION IN (12.65, 13.65)
AND    c.FK_PARENT IN (SELECT p.PK
                       FROM   BOX_ACC.T_BOX_ACCT_PORT_PROP_S p
                       WHERE  p.FK_OWNER_OBJ = 35000170.65
                       AND    p.FK_BRANCH = 21462.4
                       AND    p.FK_INSTRUMENT = 20111.4)
GROUP  BY c.FK_CONDITION, c.VALUE_FUNC
ORDER  BY c.FK_CONDITION, c.VALUE_FUNC;
