SELECT k.FK_BRANCH, k.FK_INSTRUMENT, sp.DESCRIPTION AS instrument, k.FK_PROPERTIES, t.CODE AS topic_code,
       COUNT(*) AS n_keys,
       SUM(CASE WHEN NVL(k.AMOUNT_LOC, 0) <> 0 OR NVL(k.AMOUNT_CCY, 0) <> 0 THEN 1 ELSE 0 END) AS n_keys_open,
       MAX(k.MAXDATE) AS last_maxdate

LEFT   JOIN PGT_ACT.T_PGT_ACCT_TOPICS_S t  ON t.PK  = k.FK_TOPIC
LEFT   JOIN PGT_SYS.T_PGT_SUB_PRODUCT_S sp ON sp.PK = k.FK_INSTRUMENT
WHERE  k.FK_BRANCH IN (20007.4, 21447.4)
AND    k.FK_INSTRUMENT IN (2.4, 20.4, 20092.4, 20111.4, 20213.4, 20313.4)
GROUP  BY k.FK_BRANCH, k.FK_INSTRUMENT, sp.DESCRIPTION, k.FK_PROPERTIES, t.CODE
HAVING SUM(CASE WHEN NVL(k.AMOUNT_LOC, 0) <> 0 OR NVL(k.AMOUNT_CCY, 0) <> 0 THEN 1 ELSE 0 END) > 0
ORDER  BY 3, 4, 5;

WITH used AS (
  SELECT FK_INSTRUMENT, FK_PROPERTIES, FK_TOPIC, FK_GLTA, COUNT(*) AS n
  FROM   PGT_ACT.T_PGT_ACCT_MOV_S
  WHERE  FK_BRANCH IN (20007.4, 21447.4)
  AND    REGDATE >= DATE '2026-07-01' AND REGDATE < DATE '2026-08-01'
  AND    FK_INSTRUMENT IN (2.4, 20.4, 20092.4, 20111.4, 20213.4, 20313.4)
  AND    FK_TOPIC <> 0
  GROUP  BY FK_INSTRUMENT, FK_PROPERTIES, FK_TOPIC, FK_GLTA),
nc AS (
  SELECT u.*,
         CASE WHEN NVL(u.FK_PROPERTIES, 0) = 0 THEN 'no portfolio property' ELSE 'topic not in its lists' END AS kind,
         CASE WHEN EXISTS (SELECT 1 FROM PGT_SYS.T_PGT_CROSS_ACCTCONF_S x
                           WHERE x.FK_OWNER_OBJ = 18051.4 AND x.BRANCH = 20007.4 AND x.TOPIC = u.FK_TOPIC)
              THEN 'Y' ELSE 'N' END AS in_cross_conf
  FROM   used u
  WHERE  NOT EXISTS (SELECT 1 FROM PGT_ACT.T_PGT_ACCT_LIST_TOPIC_S lt
                     WHERE lt.FK_PARENT = u.FK_PROPERTIES AND lt.FK_TOPIC = u.FK_TOPIC AND lt.FK_OWNER_OBJ = 12646.4))
SELECT sp.DESCRIPTION AS instrument, t.CODE AS topic_code, a.CODE AS account_code,
       nc.kind, nc.in_cross_conf, COUNT(*) AS combinations, SUM(nc.n) AS movements
FROM   nc
LEFT   JOIN PGT_ACT.T_PGT_ACCT_TOPICS_S  t  ON t.PK  = nc.FK_TOPIC
LEFT   JOIN PGT_ACT.T_PGT_ACCT_GLTA_S    a  ON a.PK  = nc.FK_GLTA
LEFT   JOIN PGT_SYS.T_PGT_SUB_PRODUCT_S sp ON sp.PK = nc.FK_INSTRUMENT
WHERE  NVL(a.CODE, '-') NOT IN ('INTERNAL', 'INTERNAL1')
GROUP  BY sp.DESCRIPTION, t.CODE, a.CODE, nc.kind, nc.in_cross_conf
ORDER  BY 1, 2, 3;
