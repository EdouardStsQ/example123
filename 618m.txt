SELECT 'table' AS kind, OWNER AS owner, TABLE_NAME AS name FROM ALL_TABLES
WHERE  TABLE_NAME IN ('T_PGT_BRANCH_GROUP_S', 'T_PGT_BRANCH_S')
UNION ALL
SELECT 'synonym', OWNER || ' -> ' || TABLE_OWNER, SYNONYM_NAME FROM ALL_SYNONYMS
WHERE  SYNONYM_NAME IN ('T_PGT_BRANCH_GROUP_S', 'T_PGT_BRANCH_S')
UNION ALL
SELECT 'connected as', USER, NULL FROM dual;
