-- 5. Portfolio properties of the BOX products, per group (Asia + US for calibration)
SELECT g.DESCRIPTION AS branch_group, sp.DESCRIPTION AS instrument, COUNT(*) AS n_port_props
FROM   PGT_ACT.T_PGT_ACCT_PORT_PROP_S pp
JOIN   PGT_STC.T_PGT_BRANCH_GROUP_S g  ON g.PK  = pp.FK_BRANCH
LEFT   JOIN PGT_SYS.T_PGT_SUB_PRODUCT_S sp ON sp.PK = pp.FK_INSTRUMENT
WHERE  pp.FK_OWNER_OBJ = 12646.4
AND    pp.FK_BRANCH IN (21453.4, 21457.4, 21688.45, 21447.4)
GROUP  BY g.DESCRIPTION, sp.DESCRIPTION
ORDER  BY 1, 2;

-- 6. Last accounting activity of the three branches, per instrument (last 12 months)
SELECT NVL(b.DESCRIPTION, g.DESCRIPTION) AS branch_or_group, sp.DESCRIPTION AS instrument,
       COUNT(*) AS n_movements, MIN(m.REGDATE) AS first_date, MAX(m.REGDATE) AS last_date
FROM   PGT_ACT.T_PGT_ACCT_MOV_S m
LEFT   JOIN PGT_STC.T_PGT_BRANCH_S       b  ON b.PK  = m.FK_BRANCH
LEFT   JOIN PGT_STC.T_PGT_BRANCH_GROUP_S g  ON g.PK  = m.FK_BRANCH
LEFT   JOIN PGT_SYS.T_PGT_SUB_PRODUCT_S  sp ON sp.PK = m.FK_INSTRUMENT
WHERE  m.FK_BRANCH IN (20063.4, 20081.4, 20159.4, 21453.4, 21457.4, 21688.45)
AND    m.REGDATE >= DATE '2025-11-01'
GROUP  BY NVL(b.DESCRIPTION, g.DESCRIPTION), sp.DESCRIPTION
ORDER  BY 1, 2;
