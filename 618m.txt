SELECT c.FK_INSTRUMTYPE, d.DESCRIPTION, COUNT(*) AS n_props
FROM   BOX_ACC.T_BOX_ACCT_LIST_COND_S c
LEFT   JOIN PGT_SYS.PGT_DOMAINS d ON d.PK = c.FK_INSTRUMTYPE
WHERE  c.FK_OWNER_OBJ = 35000170.65
AND    c.FK_EXTENSION = 35001514.65
AND    c.FK_CONDITION = 1.65
AND    c.FK_PARENT IN (SELECT p.PK
                       FROM   BOX_ACC.T_BOX_ACCT_PORT_PROP_S p
                       WHERE  p.FK_OWNER_OBJ = 35000170.65
                       AND    p.FK_BRANCH = 21462.4
                       AND    p.FK_INSTRUMENT = 2.4)
GROUP  BY c.FK_INSTRUMTYPE, d.DESCRIPTION
ORDER  BY 2;
