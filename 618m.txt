SELECT pp.FK_INSTRUMENT, lt.FK_PARENT AS port_prop_pk, lt.FK_EXTENSION AS list_ext, t.CODE AS topic_code,
       COUNT(*) AS n_rows, COUNT(DISTINCT lt.FK_GLTA) AS n_accounts
FROM   PGT_ACT.T_PGT_ACCT_LIST_TOPIC_S lt
JOIN   PGT_ACT.T_PGT_ACCT_PORT_PROP_S pp ON pp.PK = lt.FK_PARENT
LEFT   JOIN PGT_ACT.T_PGT_ACCT_TOPICS_S t ON t.PK = lt.FK_TOPIC
WHERE  lt.FK_OWNER_OBJ = 12646.4 AND lt.FK_EXTENSION IN (38249.4, 38329.4, 38422.4)
AND    pp.FK_OWNER_OBJ = 12646.4 AND pp.FK_BRANCH = 21447.4 AND pp.STATUS = 'Valid'
AND    pp.FK_INSTRUMENT IN (20092.4, 2.4, 20.4, 20111.4, 20213.4, 20313.4)
GROUP  BY pp.FK_INSTRUMENT, lt.FK_PARENT, lt.FK_EXTENSION, t.CODE
HAVING COUNT(DISTINCT lt.FK_GLTA) > 1
ORDER  BY 1, 2, 3, 4;



SELECT l.FK_EXTENSION AS list_ext, COUNT(*) AS n_cases
FROM  (SELECT FK_PARENT, FK_EXTENSION, FK_TOPIC FROM BOX_ACC.T_BOX_ACCT_LIST_TOPIC_S
       WHERE  FK_OWNER_OBJ = 35000170.65 AND FK_EXTENSION IN (35001510.65, 35001513.65, 35001518.65)
       GROUP  BY FK_PARENT, FK_EXTENSION, FK_TOPIC HAVING COUNT(DISTINCT FK_GLTA) > 1) l
GROUP  BY l.FK_EXTENSION;
