SELECT cp.NOM_FUNC, c.VALUE_FUNC, COUNT(*) AS n_props
FROM   BOX_ACC.T_BOX_ACCT_LIST_COND_S c
JOIN   BOX_ACC.T_BOX_CONDPAR_PROP_S cp ON cp.PK = c.FK_CONDITION
WHERE  c.FK_OWNER_OBJ = 35000170.65 AND c.FK_EXTENSION = 35001514.65
AND    AND (UPPER(cp.NOM_FUNC) LIKE '%GETCLASSOPTION%' OR UPPER(cp.NOM_FUNC) LIKE '%GETUNDERLYINGTYPEOPTION%')
AND    c.FK_PARENT IN (SELECT PK FROM BOX_ACC.T_BOX_ACCT_PORT_PROP_S
                       WHERE FK_OWNER_OBJ = 35000170.65 AND FK_BRANCH = 21462.4 AND FK_INSTRUMENT = 20111.4)
GROUP  BY cp.NOM_FUNC, c.VALUE_FUNC ORDER BY 1, 2;
