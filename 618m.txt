SELECT c.constraint_name, c.constraint_type, cc.position, cc.column_name
FROM   all_constraints c
JOIN   all_cons_columns cc ON cc.owner = c.owner AND cc.constraint_name = c.constraint_name
WHERE  c.owner = 'PGT_TRD' AND c.table_name = 'T_PGT_OBJ_SOURCE_S'
ORDER  BY c.constraint_name, cc.position;


SELECT PK, ALIASCODE, FK_PARENT, FK_OWNER_OBJ, FK_EXTENSION, FK_SOURCE
FROM   PGT_TRD.T_PGT_OBJ_SOURCE_S
WHERE  FK_PARENT = 12.22
ORDER  BY FK_SOURCE, ALIASCODE;
