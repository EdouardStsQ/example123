SELECT SEQUENCE_NAME FROM ALL_SEQUENCES
WHERE  SEQUENCE_OWNER = 'BOX_ACC' AND SEQUENCE_NAME LIKE 'SPK\_T\_BOX\_%' ESCAPE '\'
ORDER  BY 1;

SELECT GRANTED_ROLE FROM USER_ROLE_PRIVS ORDER BY 1;

SELECT TABLE_SCHEMA, TABLE_NAME,
       LISTAGG(PRIVILEGE, ',') WITHIN GROUP (ORDER BY PRIVILEGE) PRIVS
FROM  (SELECT DISTINCT TABLE_SCHEMA, TABLE_NAME, PRIVILEGE FROM ALL_TAB_PRIVS
       WHERE  TABLE_SCHEMA IN ('BOX_FE', 'BOX_ACC', 'BOX_SYS', 'PGT_SYS', 'GOM_GLB_SYS')
       AND    PRIVILEGE IN ('INSERT', 'UPDATE', 'DELETE', 'EXECUTE'))
GROUP  BY TABLE_SCHEMA, TABLE_NAME
ORDER  BY 1, 2;
