SELECT 'DEAL_DATA' step, FRONT_ID, COUNT(*) n FROM BOX_FE.T_BOX_DEAL_DATA_S
WHERE  FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318') GROUP BY FRONT_ID
UNION ALL
SELECT 'FLOW_DATA', FRONT_ID, COUNT(*) FROM BOX_FE.T_BOX_FLOW_DATA_S
WHERE  FRONT_ID IN ('76278766','79333922','79760689','80855116','80855318') GROUP BY FRONT_ID
UNION ALL
SELECT 'DATADEAL', SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1), COUNT(*) FROM BOX_FE.T_BOX_DATADEAL_S
WHERE  SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1) IN ('76278766','79333922','79760689','80855116','80855318')
GROUP  BY SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1)
UNION ALL
SELECT 'IRFINANCST', SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1), COUNT(*) FROM BOX_FE.T_BOX_IRFINANCST_S
WHERE  SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1) IN ('76278766','79333922','79760689','80855116','80855318')
GROUP  BY SUBSTR(DEAL_ID, 1, INSTR(DEAL_ID, '.') - 1)
ORDER  BY 1, 2;
