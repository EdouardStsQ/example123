SET SERVEROUTPUT ON
DECLARE
  v_max NUMBER;
  v_s   NUMBER;
  v_n   PLS_INTEGER;
  PROCEDURE fix (p_table VARCHAR2, p_seq VARCHAR2) IS
  BEGIN
    EXECUTE IMMEDIATE 'SELECT NVL(MAX(TRUNC(PK)), 0) FROM BOX_ACC.' || p_table
      || ' WHERE PK - TRUNC(PK) = 0.44' INTO v_max;
    v_n := 0;
    LOOP
      EXECUTE IMMEDIATE 'SELECT BOX_ACC.' || p_seq || '.NEXTVAL FROM DUAL' INTO v_s;
      v_n := v_n + 1;
      EXIT WHEN v_s >= v_max OR v_n > 100000;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE(p_table || ': highest .44 PK ' || v_max || ', sequence now at ' || v_s
      || ' (next value ' || (v_s + 1) || ')');
  END fix;
BEGIN
  fix('T_BOX_CONF_INSTRUM_TYPE_S', 'SPK_T_BOX_CONF_INSTRUM_TYPE_S');
  fix('T_BOX_CONF_LO_PROP_S',      'SPK_T_BOX_CONF_LO_PROP_S');
END;
/
