-- B1 the columns
SELECT TABLE_NAME, COLUMN_ID, COLUMN_NAME, DATA_TYPE
FROM   ALL_TAB_COLUMNS
WHERE  OWNER = 'BOX_FE' AND TABLE_NAME IN ('T_BOX_IRFINANCST_S', 'T_BOX_DEAL_DATA_S')
ORDER  BY 1, 2;

-- B2 rows per deal in the financial table alone
SELECT N_ROWS_PER_DEAL, COUNT(*) N_DEALS
FROM  (SELECT DEAL_ID, COUNT(*) N_ROWS_PER_DEAL
       FROM   BOX_FE.T_BOX_IRFINANCST_S
       WHERE  TRUNC(DDATE) = DATE '2026-09-25' AND FK_BRANCH = 20007.4
       GROUP  BY DEAL_ID)
GROUP  BY N_ROWS_PER_DEAL ORDER BY 1;

-- B3 what RAW received for that date, per book
SELECT BOOK, COUNT(DISTINCT FRONT_ID) N_FRONT_IDS
FROM   BOX_FE.T_BOX_RAW_DEAL_DATA_S
WHERE  TRUNC(PROCESSDATE) = DATE '2026-09-25' AND INSTRUMENT = 'IRS'
GROUP  BY BOOK ORDER BY 1;
